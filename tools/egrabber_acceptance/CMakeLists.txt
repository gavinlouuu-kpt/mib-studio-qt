cmake_minimum_required(VERSION 3.20)
project(mib_egrabber_feature_acceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(MIB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(mib_feature_control STATIC "${MIB_ROOT}/src/backend/camera/features/FeatureControl.cpp")
target_include_directories(mib_feature_control PUBLIC "${MIB_ROOT}/include")
if(MSVC)
  target_compile_options(mib_feature_control PRIVATE /W4 /WX)
else()
  target_compile_options(mib_feature_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(feature_control_test "${MIB_ROOT}/tests/camera/features/feature_control_test.cpp")
target_link_libraries(feature_control_test PRIVATE mib_feature_control)
enable_testing()
add_test(NAME egrabber.feature_control COMMAND feature_control_test)
set_tests_properties(egrabber.feature_control PROPERTIES TIMEOUT 15)

# Optional compile-only proof against an operator-provided, installed SDK.
# Does not execute the SDK, open a device, or vendor any SDK files.
set(EGRABBER_INCLUDE_DIR "" CACHE PATH "Directory containing EGrabber.h")
if(EGRABBER_INCLUDE_DIR)
  add_library(egrabber_feature_sdk_compile OBJECT "${MIB_ROOT}/tests/camera/features/sdk_compile.cpp")
  target_include_directories(egrabber_feature_sdk_compile PRIVATE "${MIB_ROOT}/include")
  target_include_directories(egrabber_feature_sdk_compile SYSTEM PRIVATE "${EGRABBER_INCLUDE_DIR}")
endif()

if(EGRABBER_INCLUDE_DIR)
  add_executable(egrabber_sdk_adapter_test "${MIB_ROOT}/tests/camera/features/sdk_adapter_test.cpp")
  target_link_libraries(egrabber_sdk_adapter_test PRIVATE mib_feature_control)
  target_include_directories(egrabber_sdk_adapter_test SYSTEM PRIVATE "${EGRABBER_INCLUDE_DIR}")
  add_test(NAME egrabber.sdk_adapter_fake COMMAND egrabber_sdk_adapter_test)
  set_tests_properties(egrabber.sdk_adapter_fake PROPERTIES TIMEOUT 15)
endif()
